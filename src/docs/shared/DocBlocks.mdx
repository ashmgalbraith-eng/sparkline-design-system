export const Callout = ({ children, tone = 'info' }) => {
  const tones = {
    info: { border: PALETTE.primary, bg: PALETTE.primaryBg },
    warning: { border: PALETTE.red, bg: PALETTE.redBg },
  };
  const t = tones[tone];
  return (
    <div style={{
      borderLeft: `3px solid ${t.border}`,
      background: t.bg,
      padding: '16px 20px',
      borderRadius: '0 8px 8px 0',
      margin: '20px 0',
      fontSize: '15px',
      color: PALETTE.text,
      lineHeight: 1.6,
    }}>
      {children}
    </div>
  );
};
