  const types = {
    Patch: { color: '#008A22', bg: '#E3F6E8' },
    Minor: { color: '#AC6502', bg: '#FCEFDC' },
    Major: { color: '#EA0606', bg: '#FDE5E5' },
  };
  const t = types[type];
  return (
    <div style={{ border: '1px solid #D6D6D6', borderRadius: '8px', padding: '20px 22px', marginBottom: '14px', background: '#fff' }}>
      <div style={{ display: 'flex', alignItems: 'center', gap: '10px', marginBottom: '10px', flexWrap: 'wrap' }}>
        <span style={{ fontFamily: 'monospace', fontSize: '13px', fontWeight: 700, color: '#000' }}>{version}</span>
        <VersionBadge label={type} color={t.color} bg={t.bg} />
        <span style={{ fontSize: '12px', color: '#9b9b9b' }}>{date}</span>
      </div>
      <div style={{ marginBottom: '10px' }}>
        {platforms.map((p) => <PlatformTag key={p} label={p} />)}
      </div>
      <ul style={{ listStyle: 'none', padding: 0, margin: 0, display: 'flex', flexDirection: 'column', gap: '6px' }}>
        {children}
      </ul>
    </div>
  );
};
